cmake_minimum_required(VERSION 3.24)
project(PaneCapture LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Pane Capture requires the Windows x64 MSVC toolchain.")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_OBJECT_PATH_MAX 180)
set(CMAKE_RC_FLAGS "${CMAKE_RC_FLAGS} /c65001")
add_compile_options($<$<COMPILE_LANGUAGE:C,CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/permissive-> $<$<COMPILE_LANGUAGE:C,CXX>:/W3>)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)

file(TO_CMAKE_PATH "${PANE_CAPTURE_SOURCES}" PANE_CAPTURE_SOURCES)
file(TO_CMAKE_PATH "${PANE_CAPTURE_ATL}" PANE_CAPTURE_ATL)
set(LING "${PANE_CAPTURE_SOURCES}/Ling")
set(CAPTURE "${PANE_CAPTURE_SOURCES}/ScreenCapture/Src")
file(GLOB LING_SOURCES CONFIGURE_DEPENDS "${LING}/src/*.cpp")
file(GLOB_RECURSE YOGA_SOURCES CONFIGURE_DEPENDS "${LING}/yoga/*.cpp")
add_library(PaneCaptureYoga STATIC ${YOGA_SOURCES})
target_include_directories(PaneCaptureYoga PUBLIC "${LING}")
add_library(PaneCaptureLing STATIC ${LING_SOURCES})
target_include_directories(PaneCaptureLing PUBLIC "${LING}" PRIVATE "${LING}/src")
target_precompile_headers(PaneCaptureLing PRIVATE "${LING}/src/pch.h")
target_link_libraries(PaneCaptureLing PUBLIC PaneCaptureYoga windowsapp runtimeobject d3d11 d2d1 dwrite
  user32 shcore dwmapi shell32 shlwapi windowscodecs ole32 oleaut32 gdi32 comdlg32 advapi32 version)

file(GLOB_RECURSE CAPTURE_SOURCES CONFIGURE_DEPENDS "${CAPTURE}/*.cpp" "${CAPTURE}/*.c")
add_executable(ScreenCapture WIN32 ${CAPTURE_SOURCES} "${CAPTURE}/Res/Resource.rc")
target_include_directories(ScreenCapture PRIVATE "${CAPTURE}" "${PANE_CAPTURE_ATL}/include")
target_link_directories(ScreenCapture PRIVATE "${PANE_CAPTURE_ATL}/lib/x64")
set_source_files_properties("${CAPTURE}/Res/Resource.rc" PROPERTIES INCLUDE_DIRECTORIES "${CAPTURE}/Res")
target_compile_definitions(ScreenCapture PRIVATE QUIRC_MAX_REGIONS=65534)
target_link_libraries(ScreenCapture PRIVATE PaneCaptureLing comctl32 imm32 version mf mfreadwrite mfplat mfuuid ntdll userenv)
target_link_options(ScreenCapture PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_LIST_DIR}/per-monitor.manifest")

option(PANE_BUILD_OCR "Build the optional OCR development target" OFF)
if(PANE_BUILD_OCR)
  find_package(OpenCV 4.14.0 EXACT CONFIG REQUIRED COMPONENTS core imgproc PATHS "${OpenCV_DIR}" NO_DEFAULT_PATH)
  set(ORT_INCLUDE "${PANE_ONNXRUNTIME_ROOT}/include")
  set(ORT_LIBRARY "${PANE_ONNXRUNTIME_ROOT}/lib/onnxruntime.lib")
  set(TINY "${PANE_CAPTURE_SOURCES}/TinyOCR")
  file(GLOB TINY_SOURCES CONFIGURE_DEPENDS "${TINY}/TinyOCR/src/*.cpp" "${TINY}/Clipper2/src/*.cpp")
  add_library(PaneTinyOcr STATIC ${TINY_SOURCES})
  target_include_directories(PaneTinyOcr PUBLIC "${TINY}/TinyOCR/include" "${TINY}/Clipper2/include" "${ORT_INCLUDE}" ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(PaneTinyOcr PUBLIC "${ORT_LIBRARY}" ${OpenCV_LIBS})
  set(READER "${PANE_CAPTURE_SOURCES}/ImageReader/ImageReader")
  file(GLOB READER_SOURCES CONFIGURE_DEPENDS "${READER}/*.cpp")
  add_executable(ImageReader WIN32 ${READER_SOURCES} "${READER}/Res/Resource.rc")
  set_source_files_properties("${READER}/Res/Resource.rc" PROPERTIES INCLUDE_DIRECTORIES "${READER}/Res")
  target_include_directories(ImageReader PRIVATE "${READER}" "${TINY}")
  target_link_libraries(ImageReader PRIVATE PaneCaptureLing PaneTinyOcr)
  target_link_options(ImageReader PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_LIST_DIR}/per-monitor.manifest")
endif()
